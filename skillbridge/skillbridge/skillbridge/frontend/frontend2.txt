
        const { useState, useEffect, useRef } = React;

        const FOOD_DATABASE = [
            { id: 1, name: 'Chicken Breast', calories: 165, protein: 31, carbs: 0, fat: 3.6, unit: '100g' },
            { id: 2, name: 'Brown Rice', calories: 112, protein: 2.6, carbs: 24, fat: 0.9, unit: '100g' },
            { id: 3, name: 'Sweet Potato', calories: 86, protein: 1.6, carbs: 20, fat: 0.1, unit: '100g' },
            { id: 4, name: 'Banana', calories: 89, protein: 1.1, carbs: 23, fat: 0.3, unit: '100g' },
            { id: 5, name: 'Eggs', calories: 155, protein: 13, carbs: 1.1, fat: 11, unit: '1 large' },
            { id: 6, name: 'Broccoli', calories: 34, protein: 2.8, carbs: 7, fat: 0.4, unit: '100g' },
            { id: 7, name: 'Salmon', calories: 208, protein: 20, carbs: 0, fat: 13, unit: '100g' },
            { id: 8, name: 'Greek Yogurt', calories: 59, protein: 10, carbs: 3.3, fat: 0.4, unit: '100g' },
            { id: 9, name: 'Almonds', calories: 579, protein: 21, carbs: 22, fat: 50, unit: '100g' },
            { id: 10, name: 'Oats', calories: 389, protein: 17, carbs: 66, fat: 6.9, unit: '100g' },
        ];

        const EXERCISE_DATABASE = [
            { id: 1, name: 'Running', caloriesPerMin: 10, duration: 30 },
            { id: 2, name: 'Cycling', caloriesPerMin: 8, duration: 45 },
            { id: 3, name: 'Swimming', caloriesPerMin: 11, duration: 30 },
            { id: 4, name: 'Weight Training', caloriesPerMin: 6, duration: 60 },
            { id: 5, name: 'Yoga', caloriesPerMin: 3, duration: 45 },
            { id: 6, name: 'Walking', caloriesPerMin: 4, duration: 30 },
            { id: 7, name: 'Jump Rope', caloriesPerMin: 12, duration: 20 },
            { id: 8, name: 'HIIT Workout', caloriesPerMin: 13, duration: 20 },
        ];

        function FitnessPro() {
            const [currentPage, setCurrentPage] = useState('dashboard');
            const [showModal, setShowModal] = useState(false);
            const [modalType, setModalType] = useState('food');
            const [foodLog, setFoodLog] = useState([]);
            const [exerciseLog, setExerciseLog] = useState([]);
            const [weight, setWeight] = useState(75);
            const [weightHistory, setWeightHistory] = useState([
                { date: 'Mon', value: 76 },
                { date: 'Tue', value: 75.5 },
                { date: 'Wed', value: 75 }
            ]);
            const [selectedFood, setSelectedFood] = useState(null);
            const [selectedExercise, setSelectedExercise] = useState(null);
            const [exerciseDuration, setExerciseDuration] = useState(30);
            const [searchQuery, setSearchQuery] = useState('');

            const calorieGoal = 2000;
            const proteinGoal = 150;
            const carbsGoal = 200;
            const fatGoal = 65;

            const totalCalories = foodLog.reduce((sum, item) => sum + item.calories, 0);
            const totalProtein = foodLog.reduce((sum, item) => sum + item.protein, 0);
            const totalCarbs = foodLog.reduce((sum, item) => sum + item.carbs, 0);
            const totalFat = foodLog.reduce((sum, item) => sum + item.fat, 0);
            const totalExerciseCalories = exerciseLog.reduce((sum, item) => sum + item.calories, 0);

            const netCalories = totalCalories - totalExerciseCalories;
            const remainingCalories = calorieGoal - netCalories;

            const addFood = () => {
                if (selectedFood) {
                    setFoodLog([...foodLog, { ...selectedFood, id: Date.now() }]);
                    setSelectedFood(null);
                    setModalType('');
                    setShowModal(false);
                }
            };

            const addExercise = () => {
                if (selectedExercise) {
                    const calories = selectedExercise.caloriesPerMin * exerciseDuration;
                    setExerciseLog([...exerciseLog, { 
                        ...selectedExercise, 
                        id: Date.now(), 
                        duration: exerciseDuration,
                        calories: Math.round(calories)
                    }]);
                    setSelectedExercise(null);
                    setExerciseDuration(30);
                    setModalType('');
                    setShowModal(false);
                }
            };

            const removeFood = (id) => {
                setFoodLog(foodLog.filter(item => item.id !== id));
            };

            const removeExercise = (id) => {
                setExerciseLog(exerciseLog.filter(item => item.id !== id));
            };

            const filteredFoods = FOOD_DATABASE.filter(food =>
                food.name.toLowerCase().includes(searchQuery.toLowerCase())
            );

            const Dashboard = () => (
                <div>
                    <div className="header-section">
                        <div className="header-content">
                            <div className="header-title">Today</div>
                            <div className="header-date">{new Date().toLocaleDateString('en-US', { weekday: 'long', month: 'short', day: 'numeric' })}</div>
                            <div className="macro-container">
                                <div className="macro-card">
                                    <div className="macro-label">Calories</div>
                                    <div className="macro-value">{Math.round(netCalories)}</div>
                                    <div className="macro-remaining">Remaining: {Math.round(remainingCalories)}</div>
                                </div>
                                <div className="macro-card">
                                    <div className="macro-label">Protein</div>
                                    <div className="macro-value">{Math.round(totalProtein)}g</div>
                                    <div className="macro-remaining">Goal: {proteinGoal}g</div>
                                </div>
                                <div className="macro-card">
                                    <div className="macro-label">Carbs</div>
                                    <div className="macro-value">{Math.round(totalCarbs)}g</div>
                                    <div className="macro-remaining">Goal: {carbsGoal}g</div>
                                </div>
                                <div className="macro-card">
                                    <div className="macro-label">Fat</div>
                                    <div className="macro-value">{Math.round(totalFat)}g</div>
                                    <div className="macro-remaining">Goal: {fatGoal}g</div>
                                </div>
                            </div>
                        </div>
                    </div>

                    <div className="dashboard-grid">
                        <div className="card">
                            <div className="card-title">
                                <span className="card-icon">🔥</span>
                                Calories
                            </div>
                            <div className="calories-circle">
                                <div className="circle-chart" style={{
                                    background: `conic-gradient(
                                        #667eea 0deg,
                                        #667eea ${(netCalories / calorieGoal) * 360}deg,
                                        #e0e0e0 ${(netCalories / calorieGoal) * 360}deg,
                                        #e0e0e0 360deg
                                    )`
                                }}>
                                    <div className="circle-inner">
                                        <div className="circle-number">{Math.round(netCalories)}</div>
                                        <div className="circle-label">/ {calorieGoal}</div>
                                    </div>
                                </div>
                            </div>
                            <button className="btn-add" onClick={() => {
                                setModalType('food');
                                setShowModal(true);
                            }}>+ Add Food</button>
                        </div>

                        <div className="card">
                            <div className="card-title">
                                <span className="card-icon">📊</span>
                                Macros Progress
                            </div>
                            <div className="progress-item">
                                <div className="progress-label">
                                    <span>Protein</span>
                                    <span>{Math.round(totalProtein)}g / {proteinGoal}g</span>
                                </div>
                                <div className="progress-bar">
                                    <div className="progress-fill" style={{ width: `${Math.min((totalProtein / proteinGoal) * 100, 100)}%` }}></div>
                                </div>
                            </div>
                            <div className="progress-item">
                                <div className="progress-label">
                                    <span>Carbs</span>
                                    <span>{Math.round(totalCarbs)}g / {carbsGoal}g</span>
                                </div>
                                <div className="progress-bar">
                                    <div className="progress-fill" style={{ width: `${Math.min((totalCarbs / carbsGoal) * 100, 100)}%` }}></div>
                                </div>
                            </div>
                            <div className="progress-item">
                                <div className="progress-label">
                                    <span>Fat</span>
                                    <span>{Math.round(totalFat)}g / {fatGoal}g</span>
                                </div>
                                <div className="progress-bar">
                                    <div className="progress-fill" style={{ width: `${Math.min((totalFat / fatGoal) * 100, 100)}%` }}></div>
                                </div>
                            </div>
                        </div>

                        <div className="card">
                            <div className="card-title">
                                <span className="card-icon">🏃</span>
                                Exercise
                            </div>
                            <div style={{ textAlign: 'center', padding: '20px 0' }}>
                                <div style={{ fontSize: '32px', fontWeight: '800', color: '#f39c12', marginBottom: '5px' }}>
                                    {totalExerciseCalories}
                                </div>
                                <div style={{ fontSize: '12px', color: '#a0aec0' }}>Calories Burned</div>
                            </div>
                            <button className="btn-add" onClick={() => {
                                setModalType('exercise');
                                setShowModal(true);
                            }}>+ Add Exercise</button>
                        </div>
                    </div>
                </div>
            );

            const FoodPage = () => (
                <div>
                    <div style={{ marginBottom: '20px' }}>
                        <button className="btn-add" onClick={() => {
                            setModalType('food');
                            setShowModal(true);
                        }}>+ Add Food</button>
                    </div>
                    <div className="card">
                        <div className="card-title">
                            <span className="card-icon">🍽️</span>
                            Food Log
                        </div>
                        {foodLog.length === 0 ? (
                            <div style={{ textAlign: 'center', padding: '40px 0', color: '#a0aec0' }}>
                                No foods logged yet. Start by adding a meal!
                            </div>
                        ) : (
                            foodLog.map(food => (
                                <div key={food.id} className="food-item">
                                    <div className="food-info">
                                        <div className="food-name">{food.name}</div>
                                        <div className="food-details">P: {food.protein}g | C: {food.carbs}g | F: {food.fat}g</div>
                                    </div>
                                    <div style={{ display: 'flex', alignItems: 'center', gap: '15px' }}>
                                        <div className="food-calories">{food.calories}</div>
                                        <button onClick={() => removeFood(food.id)} style={{ background: 'none', border: 'none', color: '#e74c3c', cursor: 'pointer', fontSize: '18px' }}>×</button>
                                    </div>
                                </div>
                            ))
                        )}
                    </div>
                </div>
            );

            const ExercisePage = () => (
                <div>
                    <div style={{ marginBottom: '20px' }}>
                        <button className="btn-add" onClick={() => {
                            setModalType('exercise');
                            setShowModal(true);
                        }}>+ Add Exercise</button>
                    </div>
                    <div className="card">
                        <div className="card-title">
                            <span className="card-icon">🏋️</span>
                            Exercise Log
                        </div>
                        {exerciseLog.length === 0 ? (
                            <div style={{ textAlign: 'center', padding: '40px 0', color: '#a0aec0' }}>
                                No exercises logged yet. Start your workout!
                            </div>
                        ) : (
                            exerciseLog.map(exercise => (
                                <div key={exercise.id} className="exercise-item">
                                    <div className="exercise-info">
                                        <div className="exercise-name">{exercise.name}</div>
                                        <div className="exercise-time">{exercise.duration} minutes</div>
                                    </div>
                                    <div style={{ display: 'flex', alignItems: 'center', gap: '15px' }}>
                                        <div className="exercise-calories">-{exercise.calories}</div>
                                        <button onClick={() => removeExercise(exercise.id)} style={{ background: 'none', border: 'none', color: '#e74c3c', cursor: 'pointer', fontSize: '18px' }}>×</button>
                                    </div>
                                </div>
                            ))
                        )}
                    </div>
                </div>
            );

            const ProgressPage = () => (
                <div>
                    <div className="card">
                        <div className="card-title">
                            <span className="card-icon">⚖️</span>
                            Weight Tracking
                        </div>
                        <div style={{ textAlign: 'center', padding: '30px 0' }}>
                            <div style={{ fontSize: '48px', fontWeight: '800', color: '#667eea', marginBottom: '10px' }}>
                                {weight} kg
                            </div>
                            <div style={{ fontSize: '12px', color: '#a0aec0', marginBottom: '20px' }}>
                                Last updated today
                            </div>
                            <input 
                                type="number" 
                                value={weight} 
                                onChange={(e) => setWeight(parseFloat(e.target.value))}
                                className="form-input"
                                style={{ marginBottom: '15px' }}
                            />
                            <button className="btn-add" onClick={() => {
                                setWeightHistory([...weightHistory, { date: new Date().toLocaleDateString('en-US', { weekday: 'short' }), value: weight }]);
                            }}>Log Weight</button>
                        </div>
                    </div>

                    <div className="card" style={{ marginTop: '20px' }}>
                        <div className="card-title">
                            <span className="card-icon">📈</span>
                            Weight History
                        </div>
                        <div className="weight-history">
                            {weightHistory.map((entry, i) => (
                                <div key={i} className="weight-entry">
                                    <div className="weight-entry-date">{entry.date}</div>
                                    <div className="weight-entry-value">{entry.value}</div>
                                </div>
                            ))}
                        </div>
                    </div>

                    <div className="stats-grid" style={{ marginTop: '20px' }}>
                        <div className="stat-box">
                            <div className="stat-value">{foodLog.length}</div>
                            <div className="stat-label">Meals Logged</div>
                        </div>
                        <div className="stat-box">
                            <div className="stat-value">{exerciseLog.length}</div>
                            <div className="stat-label">Workouts</div>
                        </div>
                        <div className="stat-box">
                            <div className="stat-value">{Math.round(totalExerciseCalories)}</div>
                            <div className="stat-label">Calories Burned</div>
                        </div>
                    </div>
                </div>
            );

            return (
                <div className="app-container">
                    <div className="sidebar">
                        <div className="sidebar-header">
                            <div className="sidebar-logo">💪</div>
                            <div style={{ fontWeight: '700', fontSize: '16px' }}>FitnessPro</div>
                        </div>
                        <ul className="sidebar-nav">
                            <li className={`nav-item ${currentPage === 'dashboard' ? 'active' : ''}`} onClick={() => setCurrentPage('dashboard')}>
                                <span className="nav-icon">📊</span>
                                <span>Dashboard</span>
                            </li>
                            <li className={`nav-item ${currentPage === 'food' ? 'active' : ''}`} onClick={() => setCurrentPage('food')}>
                                <span className="nav-icon">🍽️</span>
                                <span>Food Log</span>
                            </li>
                            <li className={`nav-item ${currentPage === 'exercise' ? 'active' : ''}`} onClick={() => setCurrentPage('exercise')}>
                                <span className="nav-icon">🏋️</span>
                                <span>Exercise</span>
                            </li>
                            <li className={`nav-item ${currentPage === 'progress' ? 'active' : ''}`} onClick={() => setCurrentPage('progress')}>
                                <span className="nav-icon">📈</span>
                                <span>Progress</span>
                            </li>
                        </ul>
                    </div>

                    <div className="main-content">
                        {currentPage === 'dashboard' && <Dashboard />}
                        {currentPage === 'food' && <FoodPage />}
                        {currentPage === 'exercise' && <ExercisePage />}
                        {currentPage === 'progress' && <ProgressPage />}
                    </div>

                    {showModal && (
                        <div className="modal-overlay" onClick={() => setShowModal(false)}>
                            <div className="modal-content" onClick={(e) => e.stopPropagation()}>
                                {modalType === 'food' && (
                                    <div>
                                        <div className="modal-title">Add Food</div>
                                        <input 
                                            type="text"
                                            placeholder="Search foods..."
                                            value={searchQuery}
                                            onChange={(e) => setSearchQuery(e.target.value)}
                                            className="search-bar"
                                        />
                                        <ul className="food-list">
                                            {filteredFoods.map(food => (
                                                <li 
                                                    key={food.id}
                                                    className="food-list-item"
                                                    onClick={() => setSelectedFood(food)}
                                                >
                                                    <div>
                                                        <div style={{ fontWeight: '600' }}>{food.name}</div>
                                                        <div style={{ fontSize: '12px' }}>{food.calories} cal / {food.unit}</div>
                                                    </div>
                                                    <div style={{ fontWeight: '600' }}>{selectedFood?.id === food.id ? '✓' : ''}</div>
                                                </li>
                                            ))}
                                        </ul>
                                        {selectedFood && (
                                            <button className="btn-add" onClick={addFood}>Add {selectedFood.name}</button>
                                        )}
                                    </div>
                                )}

                                {modalType === 'exercise' && (
                                    <div>
                                        <div className="modal-title">Add Exercise</div>
                                        <div className="form-group">
                                            <div className="form-label">Select Exercise</div>
                                            <div style={{ display: 'grid', gap: '10px' }}>
                                                {EXERCISE_DATABASE.map(exercise => (
                                                    <div
                                                        key={exercise.id}
                                                        className="food-list-item"
                                                        onClick={() => setSelectedExercise(exercise)}
                                                    >
                                                        <div>
                                                            <div style={{ fontWeight: '600' }}>{exercise.name}</div>
                                                            <div style={{ fontSize: '12px' }}>{exercise.caloriesPerMin} cal/min</div>
                                                        </div>
                                                        <div>{selectedExercise?.id === exercise.id ? '✓' : ''}</div>
                                                    </div>
                                                ))}
                                            </div>
                                        </div>

                                        {selectedExercise && (
                                            <div className="form-group">
                                                <div className="form-label">Duration: {exerciseDuration} minutes</div>
                                                <input 
                                                    type="range"
                                                    min="5"
                                                    max="120"
                                                    value={exerciseDuration}
                                                    onChange={(e) => setExerciseDuration(parseInt(e.target.value))}
                                                    className="form-input"
                                                />
                                                <div style={{ textAlign: 'center', marginTop: '15px', fontSize: '14px', color: '#667eea', fontWeight: '600' }}>
                                                    ~{Math.round(selectedExercise.caloriesPerMin * exerciseDuration)} calories
                                                </div>
                                                <button className="btn-add" onClick={addExercise}>Add Exercise</button>
                                            </div>
                                        )}
                                    </div>
                                )}
                            </div>
                        </div>
                    )}
                </div>
            );
        }

        const root = ReactDOM.createRoot(document.getElementById('root'));
        root.render(<FitnessPro />);
